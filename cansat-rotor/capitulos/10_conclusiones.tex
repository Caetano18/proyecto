\chapter{Conclusiones y recomendación}\label{cap:concl}

\begin{enumerate}
  \item \textbf{El rotor libre cumple \Req{C4} con $R=\SI{0.25}{m}$ y el drogue soltado, con
        poco margen.} La BEMT (cap.~\ref{cap:bemt}) da $V_2=\SI{6.4}{m/s}$ para el rotor solo
        ($m_2=\SI{0.49}{kg}$) y el promedio de todo el descenso es de \SI{7.1}{m/s}, contra el
        límite de \SI{8}{m/s}. El rotor frena como un disco de $\CR\approx0{,}96$. Con
        $R=\SI{0.20}{m}$ el rotor solo daría \SI{7.85}{m/s} y no cumpliría el promedio.
  \item \textbf{El límite lo ponen el área y la masa, no el perfil:} $V_2\propto\sqrt{m_2}/R$. El
        coeficiente $\CR$ explica el \SI{79}{\percent} de la varianza de $V_2$ y la densidad
        del aire el \SI{15}{\percent} (cap.~\ref{cap:sens}); paso, cuerda y número de palas, menos
        del \SI{8}{\percent} en conjunto. El perfil decide sobre todo la velocidad de giro.
  \item \textbf{El rotor gira a unas \SI{1230}{rpm}} en régimen y supera ese valor en un
        \SI{26}{\percent} durante el arranque. La fuerza centrífuga es de \SIrange{19}{22}{N} por
        pala en régimen y de \SI{30}{N} en el pico.
  \item \textbf{El paso debe ser negativo} (recomendado \ang{-3}): con paso $\ge\ang{0}$ el rotor
        no arranca solo. Con paso negativo llega al 90\,\% de su giro en \SI{1.6}{s}, tras
        \SI{22}{m} de caída, el \SI{4.3}{\percent} de la altura de liberación
        (cap.~\ref{cap:arranque}).
  \item \textbf{La estructura necesita cuatro cambios} (cap.~\ref{cap:estructura}): la pala de
        laminado simple es inestable al flameo a $0{,}96\,\Omega$, y hace falta laminado a
        $\pm\ang{45}$ con varilla de carbono en el borde de ataque (flameo a $2{,}55\,\Omega$);
        tope de batimiento elástico a \ang{+15}, con brazo $\ge\SI{10}{mm}$; ojal con al menos
        \SI{3}{mm} de ligamento; y un anillo de espuma de \SI{25}{mm} en la base para el
        aterrizaje.
  \item \textbf{Soltar el drogue exige una secuencia con enclavamiento} (cap.~\ref{cap:arq}):
        girar el anillo de traba, confirmar que giró y recién entonces liberar el drogue con el
        pasador. Una caída libre de \SIrange{0.2}{0.3}{s} entre la suelta y la apertura suma
        \SIrange{1.7}{2.5}{m/s}; si el drogue no se suelta, el rotor sigue funcionando (modo
        pendular estable, $\SI{5.8}{m/s}$).
  \item \textbf{La electrónica del equipo cubre casi todo el anexo, con cuatro hallazgos}
        (cap.~\ref{cap:sensores}): la ESP32-C3 no tiene los 19 pines que pide el diseño; \Req{E7}
        no se cumple con holgura si las cámaras graban desde la rampa (\SI{2.3}{h} contra \SI{8.6}{h}
        si se encienden al despegar); faltan la detección de separación (\Req{X3}), la memoria con
        reloj (\Req{F1}, \Req{F2}, \Req{F5}) y el sensor de rpm; y el servo de plástico es débil
        frente a \Req{M3}.
  \item \textbf{La probabilidad de cumplir \Req{C4} es del \SI{86}{\percent} como promedio de
        todo el descenso con la Arquitectura~1, y del \SI{99.8}{\percent} con la Arquitectura~2}
        (Monte Carlo, cap.~\ref{cap:arq}); como velocidad de régimen de la etapa~2 es del
        \SI{99.7}{\percent} y \SI{100}{\percent}. Son estimaciones de ingeniería que hay que
        actualizar con los ensayos.
  \item \textbf{El paracaídas sigue siendo la opción más segura} (cap.~\ref{cap:decision}); el
        autogiro gana cuando pesa el valor técnico del proyecto.
\end{enumerate}

\section*{¿Hasta qué punto es viable la autorrotación?}
\begin{itemize}
  \item \textbf{Es viable y cumple \Req{C4}}, pero el límite lo pone el área del rotor y la masa que
        vuela con él. Con $R=\SI{0.25}{m}$ y el drogue soltado la etapa~2 cae a
        \SI{6.4}{m/s}; ningún perfil, paso o torsión la baja más de \SI{0.5}{m/s}.
  \item \textbf{Es estable} (cap.~\ref{cap:estabilidad}): los modos de péndulo son estables, el
        punto de diseño está en el régimen normal de autorrotación (estado de molino) y la
        transición desde el drogue no atraviesa el anillo de vórtices. Hacen falta la pala BV y
        cuatro aletas antigiro desplegables con calaje ajustable.
  \item \textbf{El punto débil es el margen del promedio:} \SI{0.9}{m/s} con la Arquitectura~1.
        Dos palancas lo agrandan: separar el contenedor con el drogue (la masa que vuela baja a
        \SI{0.32}{kg} y $V_2$ a \SI{5.2}{m/s}, si la organización lo permite) y bajar la masa del
        CanSat dentro de $\Req{S1}$.
  \item \textbf{Perfil con espesor.} Si el contenedor es propio, la pala puede tener espesor
        (sección~\ref{sec:polares-espesor-v2}): una placa gruesa de espesor constante suma hasta \SI{0.5}{m/s} a la caída, pero un perfil con espesor NACA de
        \SI{8}{\percent} mantiene la caída (\SI{6.3}{m/s}) y baja las rpm un 8\,\%, y con ellas la carga centrífuga un 14\,\%.
  \item \textbf{Condición para seguir}: que E3 mida $\CR\ge0{,}80$ (límite inferior al 95\,\%) y que las caídas desde drone den
        menos de \SI{7.5}{m/s} con el rotor solo. Si no, pasar al paracaídas.
\end{itemize}

\begin{resultado}[Recomendación]
Avanzar con la Arquitectura~1 como base y la 2 lista para activarse: rotor de 4 palas de placa de
carbono curvada al 7\,\% ($206\times45$\,mm, laminado a $\pm\ang{45}$ con varilla en el borde de
ataque), $R=\SI{0.25}{m}$, paso de \ang{-3}, sin torsión, bisagras de batimiento con tope
elástico a \ang{+15} y pasador de Ø\,3, drogue de Ø\,\SI{25}{cm} que se suelta después de
confirmar el giro de la traba, anillo amortiguador en la base y cuatro aletas antigiro. En la
electrónica: computadora con 20 o más pines (RP2040), servo MG90S, sensor Hall de rpm, FRAM y RTC,
detección de separación con la luz del panel y la IMU, y cámaras encendidas al despegar salvo
que la medición de su corriente permita dejarlas encendidas (cap.~\ref{cap:sensores}). Mantener un
paracaídas de Ø\,\SI{70}{cm} como plan~B hasta que los ensayos confirmen $V_2$.
\end{resultado}

\section*{Decisiones abiertas}
\begin{itemize}
  \item Altura de diseño: el equipo estima la liberación a \SI{500}{m}, o sea un apogeo de
        \SI{625}{m} con la liberación al \SI{80}{\percent}; confirmarlo con el apogeo del cohete. La
        lectura alternativa (apogeo de \SI{500}{m}, liberación a \SI{400}{m}) está en la
        sección~\ref{sec:dim-tiempos}.
  \item Consultas a la organización (lista completa en el capítulo~\ref{cap:arq}):
        diámetro (\SI{90}{mm} o $95\times200$), lectura del promedio de \Req{C4}, suelta del
        drogue antes del rotor, separación del contenedor, pared de \Req{S6} en piezas no
        impresas, evento que define la separación de \Req{X3} y alcance de \Req{E7}.
  \item Uso de \texttt{PROBE\_RELEASE}, \texttt{PAYLOAD\_RELEASE} y del comando \texttt{MEC} en
        vuelo, y el estado propio de suelta del drogue.
  \item Autorización de la ANAC para soltar cargas desde un drone en los ensayos de caída
        (hacen falta al menos \SIrange{80}{100}{m}).
  \item Radio: LR2021 o RFM95W (cap.~\ref{cap:sensores}); computadora de vuelo; peso real del
        panel solar.
\end{itemize}
